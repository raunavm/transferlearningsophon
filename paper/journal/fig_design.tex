\node[lvl={c162!25}] (l162) at (5.0, 3.55) {\Kres{} classes\\[-1pt]{\footnotesize one QCD class}};
\node[lvl={c43!25}]  (l43)  at (8.6, 3.55) {\Kmid{} classes\\[-1pt]{\footnotesize topology $\times$ $b$/$c$/light}};
\node[lvl={c17!25}]  (l17)  at (12.2, 3.55) {\Kcoarse{} classes\\[-1pt]{\footnotesize topology and prong types}};
\draw[merge] (l188) -- node[above, note]{merge the}
                       node[below, note]{\LitJetClassTwoNQcdClasses{} QCD classes} (l162);
\draw[merge] (l162) -- node[above, note]{merge $g$ with $u,d,s$;}
                       node[below, note]{$e$ with $\mu$; $b$/$c$ counts} (l43);

                       node[below, note]{flavor} (l17);
\node[note, anchor=north west, text width=13.8cm] at (-0.2, 2.75)
  {Each level merges classes of the one before; every class holds only QCD jets, only decays into

   decay apart.};

\node[ctl={crand!22}] (rand) at (1.85, 1.35) {random partitions\\[-1pt]{\footnotesize \Kcoarse{} groups, random members}};

\node[note, anchor=north west, text width=13.8cm] at (-0.2, 0.62)
  {Random partitions hold the group shares of the \Kcoarse-class level, so they change what the
   classes distinguish at a fixed number of classes: many flavor boundaries but no topological order;
   the flavor-controlled partitions fix where the flavor boundaries lie.};

\caption{Design of the study. (a)~The label sets: four levels of one class hierarchy of the
\Kfull{} JetClass-II classes, with what each merge removes, and the controls, which change what the
classes distinguish without changing their number, add a mass-regression output, or remove the
particle-type and track-displacement inputs. Figure~\ref{fig:labelsets} shows the classes of selected
decays in each label set. (b)~Models with the same seed see the same training jets in the same order
from the same initial weights, so two models of one seed differ only in their labels and output
layer. Every model is evaluated with probes on its frozen features and with full fine-tuning on the
same tasks.}
\label{fig:design}
